# Logic FSM — First Working Combat Loop

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the 7-state Logic FSM so a triggered encounter runs a complete combat loop — turns cycle by CT/PULSE, actors auto-attack the nearest living opponent, win/loss is detected and exploration resumes.

**Architecture:** A single `LogicFSM` node child of `CombatScene` drives all logic states. It is actor-agnostic — it never branches on player vs enemy, only emits signals and awaits responses. For this first loop, ActionSelect is stubbed: FSM auto-selects the first living opponent and emits `action_selected` after a short delay; this is replaced with real player input when the UI FSM is built. CT advances in TurnTracker (pure static utility); damage math lives in CombatCalculator (pure static utility).

**Tech Stack:** GDScript 4, Godot 4.6, existing `CombatCharacter`/`VitalsComponent`/`StatsComponent` infrastructure, `CombatEvents` autoload event bus.

```gdscript
class_name VitalsComponent
extends Node

const AP_CAP: int = 10

var cur_hp: int = 0
var cur_mp: int = 0
var cur_ap: int = 0
var cur_ct: float = 0.0

var _max_hp: int = 0
var _max_mp: int = 0

## FUTURE: signature grows to setup(max_hp, max_mp, cur_hp, cur_mp) once the persistence
##   loop exists. cur_hp / cur_mp come from CharacterSaveData and survive across encounters
##   (dungeon attrition pillar). Also needs CombatEnd to write cur_hp / cur_mp back to
##   CharacterSaveData — without that writeback, persistence is one-directional.

func setup(max_hp: int, max_mp: int) -> void:
	_max_hp = max_hp
	_max_mp = max_mp
	cur_hp = max_hp
	cur_mp = max_mp
	cur_ap = 3
	cur_ct = 0.0
	Log.debug("VitalsComponent setup")

func reset() -> void:
	cur_ap = 3
	Log.debug("VitalsComponent reset")

func is_downed() -> bool:
	return cur_hp <= 0

func gain_ap(amount: int) -> void:
	cur_ap = min(cur_ap + amount, AP_CAP)

func take_damage(amount: int) -> void:
	cur_hp = max(0, cur_hp - amount)
	if cur_hp == 0:
		Log.info("HP reached zero")
```

## Task 6 — CombatCalculator

**Files:**
- Create: `scripts/combat/CombatCalculator.gd`

- [ ] **Step 1: Create the file**

```gdscript
class_name CombatCalculator
## Pure utility. Takes combatants in, returns resolved values out. No state, no Node base.

## Basic physical attack using the GDD damage formula:
##   damage = Power × AtkPhys / (DefPhys + 10) × Random(0.85, 1.0)
## Power is fixed at 30 for the stub basic attack (no mnemonic yet).
static func base_attack(attacker: CombatCharacter, target: CombatCharacter) -> int:
	const POWER: int = 30
	var atk: float = attacker.stats.stat(CharacterStats.Stat.AtkPhys)
	var def: float = target.stats.stat(CharacterStats.Stat.DefPhys)
	var variance: float = randf_range(0.85, 1.0)
	var result := int(POWER * atk / (def + 10.0) * variance)
	return max(1, result)
```

## Task 7 — TurnTracker

**Files:**
- Create: `scripts/combat/TurnTracker.gd`

- [ ] **Step 1: Create the file**

```gdscript
class_name TurnTracker
## Pure utility for CT/PULSE turn management. No state, no Node base.
##
## CT advances by PULSE/100 each "tick". Ticks are virtual — we loop until
## someone reaches 1.0 rather than advancing one tick per frame.
## Overflow is preserved: a fast actor's head start on the next turn is real.

static func advance_to_next_turn(actors: Array) -> CombatCharacter:
	while true:
		for actor in actors:
			if actor.vitals.is_downed():
				continue
			actor.vitals.cur_ct += actor.stats.stat(CharacterStats.Stat.Pulse) / 100.0
		var ready := _find_ready(actors)
		if ready:
			return ready
	return null  # unreachable — at least one living actor always exists when called

static func consume_ct(actor: CombatCharacter) -> void:
	actor.vitals.cur_ct -= 1.0
	Log.debug("CT consumed", actor.character_name, actor.vitals.cur_ct)

static func _find_ready(actors: Array) -> CombatCharacter:
	var best: CombatCharacter = null
	for actor in actors:
		if actor.vitals.cur_ct < 1.0:
			continue
		if best == null or actor.stats.stat(CharacterStats.Stat.Pulse) > best.stats.stat(CharacterStats.Stat.Pulse):
			best = actor
	return best
```

## Task 8 — LogicFSM

**Files:**
- Create: `scripts/combat/LogicFSM.gd`

- [ ] **Step 1: Create the file**

```gdscript
class_name LogicFSM
extends Node

enum State {
	NONE,
	COMBAT_START,
	TURN_START,
	ACTION_SELECT,
	ACTION_EXECUTION,
	REACTION_SCAN,
	TURN_END,
	COMBAT_END
}

var _state: State = State.NONE
var _actors: Array = []
var _active_actor: CombatCharacter = null
var _pending_target: CombatCharacter = null

func start(actors: Array) -> void:
	_actors = actors
	Log.info("LogicFSM starting with", _actors.size(), "actors")
	_transition(State.COMBAT_START)

func _transition(next: State) -> void:
	Log.info("LogicFSM", State.keys()[_state], "→", State.keys()[next])
	_state = next
	match next:
		State.COMBAT_START:     _enter_combat_start()
		State.TURN_START:       _enter_turn_start()
		State.ACTION_SELECT:    _enter_action_select()
		State.ACTION_EXECUTION: _enter_action_execution()
		State.REACTION_SCAN:    _enter_reaction_scan()
		State.TURN_END:         _enter_turn_end()
		State.COMBAT_END:       _enter_combat_end()

func _enter_combat_start() -> void:
	CombatEvents.combat_started.emit()
	_transition(State.TURN_START)

func _enter_turn_start() -> void:
	_active_actor = TurnTracker.advance_to_next_turn(_actors)
	_active_actor.vitals.gain_ap(1)
	_active_actor.tokens.tick_turn_start()
	Log.info("TurnStart", _active_actor.character_name, "AP:", _active_actor.vitals.cur_ap)
	CombatEvents.turn_started.emit(_active_actor)
	_transition(State.ACTION_SELECT)

func _enter_action_select() -> void:
	# STUB: FSM auto-selects the first living opponent and attacks after a short delay.
	# When the UI FSM is built, replace this entire function body with:
	#   CombatEvents.please_select_action.emit(_active_actor)
	#   var result = await CombatEvents.action_selected
	#   _pending_target = result[1]
	#   _transition(State.ACTION_EXECUTION)
	CombatEvents.please_select_action.emit(_active_actor)
	await get_tree().create_timer(0.4).timeout
	_pending_target = _first_living_opponent(_active_actor)
	if _pending_target == null:
		Log.warn("ActionSelect — no living opponent found, skipping to TurnEnd")
		_transition(State.TURN_END)
		return
	Log.info("ActionSelect", _active_actor.character_name, "→", _pending_target.character_name)
	CombatEvents.action_selected.emit(null, _pending_target)
	_transition(State.ACTION_EXECUTION)

func _enter_action_execution() -> void:
	var damage := CombatCalculator.base_attack(_active_actor, _pending_target)
	_pending_target.vitals.take_damage(damage)
	Log.info("ActionExecution", _active_actor.character_name, "hits", _pending_target.character_name, "for", damage)
	CombatEvents.action_executed.emit(_active_actor, _pending_target, damage)
	if _pending_target.vitals.is_downed():
		Log.info("actor downed", _pending_target.character_name)
		CombatEvents.actor_downed.emit(_pending_target)
	_transition(State.REACTION_SCAN)

func _enter_reaction_scan() -> void:
	# STUB: reactions not yet implemented
	_transition(State.TURN_END)

func _enter_turn_end() -> void:
	TurnTracker.consume_ct(_active_actor)
	CombatEvents.turn_ended.emit(_active_actor)
	var enemies_alive := _actors.any(func(a): return not a.is_player_controlled and not a.vitals.is_downed())
	var players_alive := _actors.any(func(a): return a.is_player_controlled and not a.vitals.is_downed())
	if not enemies_alive or not players_alive:
		_transition(State.COMBAT_END)
	else:
		_transition(State.TURN_START)

func _enter_combat_end() -> void:
	var player_won := _actors.any(func(a): return a.is_player_controlled and not a.vitals.is_downed())
	Log.info("CombatEnd — player_won:", player_won)
	CombatEvents.combat_ended.emit(player_won)

func _first_living_opponent(actor: CombatCharacter) -> CombatCharacter:
	for a in _actors:
		if a.is_player_controlled != actor.is_player_controlled and not a.vitals.is_downed():
			return a
	return null
```
